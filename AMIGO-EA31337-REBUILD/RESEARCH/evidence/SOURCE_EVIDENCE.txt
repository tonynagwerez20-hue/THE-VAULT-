412:       double st_d = test_stops[k];
413:       double act_risk = 0.0; string err = "";
414:       double raw_l = (ts>0.0 && tv_loss_risk>0.0) ? (2.50) / ((st_d / ts) * tv_loss_risk) : 0.0;
415:       double calc_lots = ComputeLotSize(g_symbol, 2000.0, 2000.0 - st_d, 0.5, act_risk, err);
416:       double ocp_min_loss = 0.0;
417:       if(v_min_valid<=0.0 || !OrderCalcProfit(ORDER_TYPE_BUY, g_symbol, v_min_valid, 2000.0, 2000.0 - st_d, ocp_min_loss))
418:       {
419:          ocp_ok = false;
420:       }
421:       double estimate_min_loss=(ts>0.0 && tv_loss_risk>0.0 && v_min_valid>0.0) ?
422:                                (st_d/ts)*tv_loss_risk*v_min_valid : 0.0;
423:       PrintFormat("[P22_AUDIT][TASK4] stop=%.2f risk_b=2.50 raw=%.5f norm=%.2f min_volume=%.5f estimated_min_loss=%.2f ocp_min_loss=%.2f dec=%s",
424:                   st_d, raw_l, calc_lots, v_min_valid, estimate_min_loss,
425:                   MathAbs(ocp_min_loss), (calc_lots>0)?"APPROVED":"REJECTED");
426:    }
427:    if(!ocp_ok)
428:    {
429:       LogMsg(LOG_ERROR, "INIT", "OrderCalcProfit audit experienced call failures; EA initialization blocked.");
430:       return INIT_FAILED;
431:    }
432:
433:    return INIT_SUCCEEDED;
434: }

445: //+------------------------------------------------------------------+
446: void OnTick()
447: {
448:    if(!PositionStateOwnerMatches(g_cfg)) return;
449:    // Lifecycle/persistence ambiguity blocks new opportunities. Timer-driven
450:    // management continues so existing positions can still be reconciled.
451:    if(!PositionManagerAdmissionAllowed()) return;
452:    // Position recovery and timer-driven management can still operate when
453:    // the account risk ledger is unavailable, but no entry analysis may run.
454:    if(!g_risk_admission_ready) return;
455:
456:    //--- Diagnostic Point 1: Throttled OnTick Heartbeat (every 60 seconds)
457:    static datetime s_last_tick_diag = 0;
458:    if(TimeCurrent() - s_last_tick_diag >= 60)
459:    {
460:       double cur_bid = SymbolInfoDouble(g_symbol, SYMBOL_BID);
461:       double cur_ask = SymbolInfoDouble(g_symbol, SYMBOL_ASK);
462:       datetime cur_bar = iTime(g_symbol, g_cfg.tf_exec, 0);
463:       PrintFormat("[RUNTIME_DIAG][OnTick] server_time=%s sym=%s bid=%.5f ask=%.5f cur_bar=%s last_bar=%s",
464:                   TimeToString(TimeCurrent(), TIME_DATE|TIME_MINUTES|TIME_SECONDS),
465:                   g_symbol, cur_bid, cur_ask,
466:                   TimeToString(cur_bar, TIME_DATE|TIME_MINUTES|TIME_SECONDS),
467:                   TimeToString(g_last_bar_time, TIME_DATE|TIME_MINUTES|TIME_SECONDS));
468:       s_last_tick_diag = TimeCurrent();
469:    }
470:
471:    //--- Risk update every tick
472:    bool risk_update_ok=RiskTick();
473:    if(!risk_update_ok && RiskAccountLockWasContended())
474:       return; // another chart currently owns account serialization; retry on a later tick
475:    if(risk_update_ok) risk_update_ok=RiskControllerUpdate();
476:    if(!risk_update_ok && RiskAccountLockWasContended())
477:       return; // transient account-wide contention is not ledger corruption
478:    if(!risk_update_ok)
479:    {
480:       PositionManagerLatchStateFault("risk ledger update failed; new admission blocked");
481:       return;
482:    }
483:
484:    //--- Feed live tick observation into native order-flow engines
485:    double last_bid = SymbolInfoDouble(g_symbol, SYMBOL_BID);
486:    double last_ask = SymbolInfoDouble(g_symbol, SYMBOL_ASK);
487:    double last_price = (last_bid > 0.0) ? last_bid : _Point;
488:    long last_vol = SymbolInfoInteger(g_symbol, SYMBOL_VOLUME);
489:    double tick_vol = (last_vol > 0) ? (double)last_vol : 1.0;
490:    if(InpEnableTickPressure)
491:    {
492:       MqlDateTime tick_dt;
493:       TimeToStruct(TimeCurrent(),tick_dt);
494:       bool tick_in_session=(tick_dt.hour>=g_cfg.session_start_hour &&
495:                             tick_dt.hour<g_cfg.session_end_hour);
496:       int tick_session_tag=(int)((TimeCurrent()-g_cfg.session_start_hour*3600)/86400);
497:       bool tick_session_started=tick_in_session &&
498:          (!g_flow_session_initialized || !g_flow_was_in_session ||
499:           g_flow_last_session_open_epoch!=tick_session_tag);
500:       if(tick_session_started)
501:       {
502:          // These are streaming diagnostic accumulators, partitioned at the
503:          // observed session boundary before the first tick is ingested.
504:          g_footprint_engine.Reset();
505:          g_vwap_engine.Reset();
506:          g_flow_last_session_open_epoch=tick_session_tag;
507:       }
508:       g_flow_session_initialized=true;
509:       g_flow_was_in_session=tick_in_session;
510:       g_footprint_engine.AddTick(last_price, tick_vol, 0);
511:       g_vwap_engine.AddObservation(last_price, tick_vol);
512:    }
513:
514:    //--- External context refresh (throttled to one read every 5 seconds)
515:    if(TimeCurrent() - g_last_ext_read >= 5)
516:    {
517:       string err;
518:       ExternalContext tmp;
519:       ZeroMemory(tmp);
520:       if(ReadExternalContext(g_cfg.bridge_inbox, tmp, err,g_cfg.stale_threshold_s))
521:       {
522:          g_ext = tmp;
523:          g_last_ext_read = TimeCurrent();
524:       }
525:       else
526:       {
527:          // Preserve non-news context for diagnostics, but never preserve a
528:          // prior news-clear state after a failed current read.
529:          g_ext.news_valid=false;
530:          g_ext.data_quality |= DQ_STALE_EXTERNAL;
531:          // Bound repeated file-open attempts when the bridge is unavailable.
532:          // The news filter remains fail-closed until a later read succeeds.
533:          g_last_ext_read = TimeCurrent();
534:       }
535:    }
536:    // Re-evaluate the retained snapshot on every tick. Read failures and
537:    // expiry both make NEWS FILTER unavailable instead of silently quiet.
538:    // ExternalContext.timestamp and news_observed_ts are UTC epoch values
539:    // produced by Python datetime.now(timezone.utc); compare them in UTC.
540:    datetime ext_now=TimeGMT();
541:    if(g_ext.timestamp<=0 || ext_now<g_ext.timestamp ||
542:       (ext_now-g_ext.timestamp)>g_cfg.stale_threshold_s)
543:    {
544:       g_ext.news_valid=false;
545:       g_ext.data_quality |= DQ_STALE_EXTERNAL;
546:    }
547:
548:    //--- Decision only on new closed bar
549:    datetime bar_time = iTime(g_symbol, g_cfg.tf_exec, 0);
550:    datetime closed_bar_time = iTime(g_symbol, g_cfg.tf_exec, 1);
551:    // Reject unavailable/inconsistent boundaries. A durable scoped reservation
552:    // is written before analysis; repeated and backward opportunities fail closed.
553:    if(bar_time<=0 || closed_bar_time<=0 || closed_bar_time>=bar_time ||
554:       (g_last_bar_time>0 && bar_time<g_last_bar_time)) return;
555:    if(bar_time != g_last_bar_time)
556:    {
557:       PrintFormat("[RUNTIME_DIAG][NEW_BAR] server_time=%s bar_time=%s prev_last_bar=%s iTime_ok=%s",
558:                   TimeToString(TimeCurrent(), TIME_DATE|TIME_MINUTES|TIME_SECONDS),
559:                   TimeToString(bar_time, TIME_DATE|TIME_MINUTES|TIME_SECONDS),
560:                   TimeToString(g_last_bar_time, TIME_DATE|TIME_MINUTES|TIME_SECONDS),
561:                   (bar_time > 0) ? "YES" : "NO");
562:    }
563:    if(bar_time == g_last_bar_time) return;
564:    long opportunity_id=0;
565:    if(!RiskBeginOpportunity(bar_time,opportunity_id))
566:    {
567:       LogMsg(LOG_WARN,"OPPORTUNITY","Duplicate, stale or ambiguous opportunity identity; admission blocked.");
568:       return;
569:    }
570:    g_last_bar_time = bar_time;
571:
572:    int outcome=OnClosedBar(opportunity_id);
573:    // Verify journal completion while the account admission lock remains held.
574:    // Failure leaves the persisted opportunity RESERVED for reconciliation.
575:    if(!ReconFinishOpportunity(opportunity_id,outcome,g_cfg))
576:       PositionManagerLatchStateFault("execution journal cannot retire this opportunity");
577:    else if(!RiskCompleteOpportunity(outcome))
578:       PositionManagerLatchStateFault("opportunity outcome could not be durably committed");
579: }

581: //+------------------------------------------------------------------+
582: int OnClosedBar(const long opportunity_id)
583: {
584:    datetime decision_boundary=iTime(g_symbol,g_cfg.tf_exec,0);
585:    datetime closed_bar_time=iTime(g_symbol,g_cfg.tf_exec,1);
586:    if(opportunity_id<=0 || decision_boundary<=0 || closed_bar_time<=0 ||
587:       decision_boundary!=g_last_bar_time || closed_bar_time>=decision_boundary)
588:       return RISK_OP_REJECTED;
589:    //--- Diagnostic Point 3: OnClosedBar Entry & Session Audit
590:    MqlDateTime dt;
591:    TimeToStruct(TimeCurrent(), dt);
592:    bool session_pass = (dt.hour >= g_cfg.session_start_hour && dt.hour < g_cfg.session_end_hour);
593:    PrintFormat("[RUNTIME_DIAG][OnClosedBar] server_time=%s sym=%s tf=%d closed_bar=%s hour=%d session=[%d-%d] pass=%s",
594:                TimeToString(TimeCurrent(), TIME_DATE|TIME_MINUTES|TIME_SECONDS),
595:                g_symbol, (int)g_cfg.tf_exec,
596:                TimeToString(iTime(g_symbol, g_cfg.tf_exec, 0), TIME_DATE|TIME_MINUTES|TIME_SECONDS),
597:                dt.hour, g_cfg.session_start_hour, g_cfg.session_end_hour,
598:                session_pass ? "YES" : "NO");
599:
600:    // Replay the full closed interval, including ticks coalesced by OnTick.
601:    // A missed interval/restart resets warm-up rather than stitching a gap.
602:    if(g_tib_last_boundary!=closed_bar_time) g_tib.Reset(InpTIBThreshold);
603:    MqlTick tib_ticks[];
604:    int tib_count=CopyTicksRange(g_symbol,tib_ticks,COPY_TICKS_ALL,
605:                                (ulong)closed_bar_time*1000,(ulong)decision_boundary*1000-1);
606:    if(tib_count<=0) { g_tib.Reset(InpTIBThreshold); return RISK_OP_REJECTED; }
607:    for(int i=0;i<tib_count;i++)
608:    {
609:       if(tib_ticks[i].time_msc<(long)closed_bar_time*1000 ||
610:          tib_ticks[i].time_msc>=(long)decision_boundary*1000 ||
611:          !MathIsValidNumber(tib_ticks[i].bid) || !MathIsValidNumber(tib_ticks[i].ask) ||
612:          tib_ticks[i].bid<=0 || tib_ticks[i].ask<tib_ticks[i].bid ||
613:          (i>0 && tib_ticks[i].time_msc<tib_ticks[i-1].time_msc))
614:       { g_tib.Reset(InpTIBThreshold); g_tib_last_boundary=0; return RISK_OP_REJECTED; }
615:       g_tib.Add(tib_ticks[i].bid,tib_ticks[i].ask,tib_ticks[i].time_msc);
616:    }
617:    g_tib_last_boundary=decision_boundary;
618:    g_tib_decision_ready=g_tib.ClosedBefore((long)decision_boundary*1000,g_tib_decision);
619:    if(!g_tib_decision_ready || g_tib_decision.close_msc<(long)closed_bar_time*1000)
620:       return RISK_OP_NO_TRADE;
621:
622:    //--- 0) Session hour gate
623:    if(dt.hour < g_cfg.session_start_hour || dt.hour >= g_cfg.session_end_hour)
624:    {
625:       return RISK_OP_REJECTED;
626:    }
627:
628:    // NEWS FILTER is admission-only and runs before any session/profile,
629:    // feature, regime, or strategy state mutation. A failed or stale provider
630:    // observation is not equivalent to a quiet calendar.
631:    string news_reason="";
632:    if(!ExternalNewsFilterAllows(g_ext,g_cfg.news_mode,TimeGMT(),
633:                                g_cfg.stale_threshold_s,news_reason))
634:    {
635:       LogMsg(LOG_WARN,"NEWS_FILTER",news_reason);
636:       return RISK_OP_REJECTED;
637:    }
638:
639:    //--- 1) Snapshot
640:    MarketSnapshot candidate_snap;
641:    FeatureSnapshot candidate_feat;
642:    AMIGOFeatureState candidate_feature_state = g_amigo_feature_state;
643:    MqlRates candidate_feature_rates[];
644:    RegimePersist candidate_regime = g_regime_state;
645:    ZeroMemory(candidate_snap);
646:    ZeroMemory(candidate_feat);
647:    bool snap_res = BuildMarketSnapshot(g_symbol,candidate_snap,decision_boundary,
648:                                        closed_bar_time,opportunity_id,g_cfg.tf_exec);
649:    PrintFormat("[RUNTIME_DIAG][SNAPSHOT] server_time=%s success=%s bid=%.5f ask=%.5f dq=%d",
650:                TimeToString(TimeCurrent(), TIME_DATE|TIME_MINUTES|TIME_SECONDS),
651:                snap_res ? "YES" : "NO", candidate_snap.bid, candidate_snap.ask, candidate_snap.data_quality);
652:    if(!snap_res)
653:    {
654:       LogMsg(LOG_WARN, "BAR", "snapshot build failed");
655:       return RISK_OP_REJECTED;
656:    }
657:    if(candidate_snap.snapshot_id!=opportunity_id ||
658:       candidate_snap.decision_boundary!=decision_boundary ||
659:       candidate_snap.closed_bar_time!=closed_bar_time)
660:       return RISK_OP_REJECTED;
661:    candidate_snap.external_max_age_s=g_cfg.stale_threshold_s;
662:
663:    //--- 2) Features
664:    bool feat_res = BuildFeatureVector(g_symbol, g_cfg, candidate_feat, candidate_feature_state,
665:                                       candidate_feature_rates);
666:    PrintFormat("[RUNTIME_DIAG][FEATURES] server_time=%s success=%s ts=%s atr14=%.5f fusion=%.2f",
667:                TimeToString(TimeCurrent(), TIME_DATE|TIME_MINUTES|TIME_SECONDS),
668:                feat_res ? "YES" : "NO",
669:                TimeToString(candidate_feat.timestamp, TIME_DATE|TIME_MINUTES|TIME_SECONDS),
670:                candidate_feat.atr14, candidate_feat.fusion);
671:    if(!feat_res)
672:    {
673:       LogMsg(LOG_WARN, "BAR", "feature build failed");
674:       return RISK_OP_REJECTED;
675:    }
676:    if(candidate_feat.timestamp!=closed_bar_time ||
677:       iTime(g_symbol,g_cfg.tf_exec,0)!=decision_boundary ||
678:       iTime(g_symbol,g_cfg.tf_exec,1)!=closed_bar_time)
679:       return RISK_OP_REJECTED;
680:    // Bind the exported close to the exact validated history read that produced
681:    // the feature snapshot. Do not re-read iClose during later publication.
682:    int validated_feature_count=ArraySize(candidate_feature_rates);
683:    if(validated_feature_count<=0 || !ArrayGetAsSeries(candidate_feature_rates) ||
684:       candidate_feature_rates[0].time!=closed_bar_time ||
685:       !MathIsValidNumber(candidate_feature_rates[0].close) ||
686:       candidate_feature_rates[0].close<=0.0)
687:       return RISK_OP_REJECTED;
688:    candidate_snap.closed_bar_close=candidate_feature_rates[0].close;
689:    candidate_feat.snapshot_id=opportunity_id;
690:    candidate_feat.decision_boundary=decision_boundary;
691:    candidate_feat.closed_bar_time=closed_bar_time;
692:    candidate_feat.symbol=g_symbol;
693:    candidate_feat.timeframe=(int)g_cfg.tf_exec;
694:
695:    //--- Diagnostic Top-Down Multi-Timeframe Context (Section 7)
696:    TopDownContext td;
697:    if(!EvaluateTopDownContext(g_symbol,g_cfg.atr_period,decision_boundary,td) ||
698:       td.asof_boundary!=decision_boundary ||
699:       iTime(g_symbol,g_cfg.tf_exec,0)!=decision_boundary ||
700:       iTime(g_symbol,g_cfg.tf_exec,1)!=closed_bar_time)
701:    {
702:       LogMsg(LOG_WARN,"TOPDOWN","INCOMPLETE_OR_MIXED_TIME_HISTORY");
703:       return RISK_OP_REJECTED;
704:    }
705:    LogMsg(LOG_INFO, "TOPDOWN_DIAG", StringFormat("ts=%s d1=%d h4=%d h1=%d m15=%d aligned=%s score=%.2f summary=%s",
706:           TimeToString(TimeCurrent(), TIME_DATE|TIME_MINUTES),
707:           td.d1_dir, td.h4_dir, td.h1_dir, td.m15_dir,
708:           td.aligned ? "TRUE" : "FALSE", td.alignment_score, td.summary));
709:
710:    //--- 4) Multi-Timeframe Regime
711:    FeatureSnapshot decision_feat = candidate_feat;
712:    if(!g_cfg.tick_pressure_enabled)
713:    {
714:       // Keep raw proxy values for diagnostics, but prevent them influencing
715:       // authoritative regime/strategy scores unless explicitly enabled.
716:       decision_feat.fusion = 0.0;
717:       decision_feat.pressure_state = 0;
718:       decision_feat.bull_flip = false;
719:       decision_feat.bear_flip = false;
720:       decision_feat.surge = false;
721:       decision_feat.delta_divergence = 0.0;
722:       decision_feat.divergence_bull = 0.0;
723:       decision_feat.divergence_bear = 0.0;
724:    }
725:    RegimeState rs = EvaluateRegimeWithState(decision_feat, td, g_cfg, candidate_regime);
726:    if(rs.snapshot_id!=opportunity_id || decision_feat.snapshot_id!=candidate_snap.snapshot_id)
727:       return RISK_OP_REJECTED;
728:
729:    //--- 5) Strategy scores
730:    StrategyScores ss = ScoreStrategies(decision_feat, g_ext, g_cfg);
731:
732:    //--- 6) ML gate (no-op when disabled)
733:    MLGateResult ml = MLGateEvaluate(candidate_feat, ss, g_cfg);
734:    if(ml.available && !ml.pass)
735:    {
736:       long did = opportunity_id;
737:       LogDecision(did, g_symbol, ACTION_NO_TRADE, rs.regime, HYP_NONE,
738:                   MathMax(ss.s_long, ss.s_short), "ML_VETO");
739:       return RISK_OP_REJECTED;
740:    }
741:
742:    //--- 7) Decision gate
743:    DecisionResult dr = Decide(ss, rs, g_ext, g_cfg,
744:                               g_ext.data_quality | candidate_snap.data_quality);
745:    if(dr.snapshot_id!=opportunity_id ||
746:       iTime(g_symbol,g_cfg.tf_exec,0)!=decision_boundary ||
747:       iTime(g_symbol,g_cfg.tf_exec,1)!=closed_bar_time)
748:       return RISK_OP_REJECTED;
749:    if(dr.action==ACTION_TRADE && dr.hypothesis==HYP_CONTINUATION &&
750:       dr.direction*g_tib_decision.imbalance<=0)
751:    {
752:       dr.action=ACTION_NO_TRADE;
753:       dr.reason="TIB_OPPOSES_CONTINUATION";
754:    }
755:    long decision_id = opportunity_id;
756:    LogDecision(decision_id, g_symbol, dr.action, rs.regime, dr.hypothesis,
757:                dr.best_score, dr.reason);
758:
759:    LogFeatureDiagnostics(candidate_feat.timestamp, g_symbol, g_cfg.tf_exec,
760:                          (int)rs.regime, candidate_feat.structure_dir, candidate_feat.fusion,
761:                          candidate_feat.bull_flip, candidate_feat.bear_flip, candidate_feat.surge,
762:                          candidate_feat.sweep_reject, ss.long_cont, ss.short_cont,
763:                          ss.long_mr, ss.short_mr, ss.s_long, ss.s_short,
764:                          dr.best_score, dr.margin, g_cfg.score_threshold,
765:                          g_cfg.score_margin, dr.action, dr.reason);
766:
767:    if(dr.action != ACTION_TRADE)
768:    {
769:       // A fully assembled and evaluated no-trade opportunity may advance accepted
770:       // strategy history. Invalid/failed paths above leave globals untouched.
771:       CommitFlowOpportunityDiagnostics(candidate_feature_rates,decision_boundary,candidate_feat.atr14);
772:       g_snap = candidate_snap;
773:       g_feat = candidate_feat;
774:       g_amigo_feature_state = candidate_feature_state;
775:       g_regime_state = candidate_regime;
776:       WriteSnapshotForExternal(g_cfg.bridge_outbox, candidate_snap, candidate_feat);
777:       return RISK_OP_NO_TRADE;
778:    }
779:
780:    //--- M1 Microstructure Execution Filter (Workstream B):
781:    // For SCALPING horizon (or M1/M5 execution), check M1 structure for immediate adverse impulse.
782:    if(g_cfg.tf_exec == PERIOD_M1 || (g_cfg.tf_exec == PERIOD_M5 && td.m1_dir != 0 && td.m1_dir != dr.direction))
783:    {
784:       if(td.m1_dir != 0 && td.m1_dir != dr.direction)
785:       {
786:          LogMsg(LOG_INFO, "EXEC_M1_VETO", StringFormat("M1 adverse microstructure veto: m1_dir=%d trade_dir=%d", td.m1_dir, dr.direction));
787:          return RISK_OP_REJECTED;
788:       }
789:    }
790:
791:    ENUM_HORIZON selected_horizon=(ENUM_HORIZON)0;
792:    if(!SelectApprovedHorizon(td,dr,rs.regime,decision_boundary,selected_horizon))
793:    {
794:       LogDecision(decision_id,g_symbol,ACTION_NO_TRADE,rs.regime,dr.hypothesis,
795:                   dr.best_score,"NO_ELIGIBLE_HORIZON");
796:       return RISK_OP_NO_TRADE;
797:    }
798:    dr.horizon=selected_horizon;
799:    string horizon_news_reason="";
800:    if(!AmigoNewsHorizonAllows(g_ext,g_cfg.news_mode,dr.horizon,TimeGMT(),
801:                               g_cfg.stale_threshold_s,horizon_news_reason))
802:    {
803:       LogDecision(decision_id,g_symbol,ACTION_NO_TRADE,rs.regime,dr.hypothesis,
804:                   dr.best_score,horizon_news_reason);
805:       return RISK_OP_REJECTED;
806:    }
807:    LogMsg(LOG_INFO,"NEWS_HORIZON",StringFormat("id=%I64d horizon=%d event_utc=%I64d reason=%s",
808:           decision_id,(int)dr.horizon,(long)g_ext.news_event_ts,horizon_news_reason));
809:    LogMsg(LOG_INFO,"FROZEN_FOOTPRINT",StringFormat("id=%I64d horizon=%d strategy=%d available=%s pressure=%.4f poc=%.5f concentration=%.4f observations=%d persistent=%s bull_div=%s bear_div=%s provenance=MT5_TICK_PROXY",
810:       decision_id,(int)dr.horizon,(int)dr.hypothesis,decision_feat.footprint_available?"TRUE":"FALSE",
811:       decision_feat.footprint_pressure,decision_feat.footprint_poc,decision_feat.footprint_concentration,
812:       decision_feat.footprint_observations,decision_feat.footprint_persistent?"TRUE":"FALSE",
813:       decision_feat.footprint_divergence_bull?"TRUE":"FALSE",decision_feat.footprint_divergence_bear?"TRUE":"FALSE"));
814:
815:    string proxy_reason;
816:    if(!HorizonProxyAgreement(decision_feat,td,dr,rs.regime,g_cfg,decision_boundary,proxy_reason))
817:    {
818:       LogDecision(decision_id,g_symbol,ACTION_NO_TRADE,rs.regime,dr.hypothesis,dr.best_score,proxy_reason);
819:       return RISK_OP_NO_TRADE;
820:    }
821:    LogMsg(LOG_INFO,"HORIZON_PROXY",StringFormat("id=%I64d horizon=%d hypothesis=%d pressure=%.3f bull_div=%.3f bear_div=%.3f provenance=%s streaming_footprint=DIAGNOSTIC broker_depth=UNAVAILABLE",
822:       decision_id,(int)dr.horizon,(int)dr.hypothesis,decision_feat.fusion,
823:       decision_feat.divergence_bull,decision_feat.divergence_bear,proxy_reason));
824:
825:
826:    //--- 8) Risk gate
827:    string risk_reason;
828:    if(!RiskAllows(g_cfg, risk_reason))
829:    {
830:       LogMsg(LOG_WARN, "RISK", risk_reason);
831:       return RISK_OP_REJECTED;
